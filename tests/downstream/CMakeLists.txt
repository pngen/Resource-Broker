cmake_minimum_required(VERSION 3.26)
project(rb_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ResourceBroker CONFIG REQUIRED)
add_executable(rb_consumer consumer.cpp)
target_link_libraries(rb_consumer PRIVATE ResourceBroker::resourcebroker)
if(MSVC)
  target_compile_options(rb_consumer PRIVATE /W4 /WX /EHsc)
else()
  target_compile_options(rb_consumer PRIVATE -Wall -Wextra -Werror)
endif()
